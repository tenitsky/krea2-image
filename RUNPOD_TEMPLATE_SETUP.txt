# 4. STORAGE
# ------------------------------------------------------------------------------

Container disk:           40 GB
# A Global Volume stages each download on the container disk (the FP8 model is
# 12.2 GB, bf16 24.5 GB) and keeps ComfyUI there. With a Network Volume, 10 GB is
# enough - but the template has one value, so 40 GB covers both.

Volume disk:              0 GB
Volume mount path:        /workspace
# Each deployer attaches their own Network Volume (40 GB) or Global Volume at
# /workspace when deploying. Setup detects which one and says so in the log.
